import kh2lab
roo = kh2lab.connect()
LO, HI = 0x01ADE000, 0x01ADF000
N = (HI - LO) // 4
Y = 0x01ADEAC4

def play(fps):
    kh2lab.start(roo, fps)
    roo.frame_advance(6)
    roo.input_set('LLeft', 'LDown'); roo.frame_advance(8); roo.input_release()
    roo.input_set('Cross'); roo.frame_advance(2); roo.input_release()

        blk[t] = np.frombuffer(roo.read_bytes(LO, HI - LO), dtype=np.uint32)
        tk[t] = roo.read(kh2lab.TICKS)

    np.savez(kh2lab.SCRATCH / f'ballobj-{fps}.npz', lo=LO, block=blk, ticks=tk)
    with np.errstate(invalid='ignore', over='ignore'):

    y = f[:, (Y - LO) // 4]
    rest = y[0]
    h = rest - y
    tmax = int(np.argmax(h))

    land = next((t for t in range(tmax + 1, 180) if h[t] <= 0.5), None)
    air = None if land is None else land - lift

    print('  height per vsync from lift: ' + ' '.join(f'{v:.0f}' for v in h[lift:lift + 110]))
