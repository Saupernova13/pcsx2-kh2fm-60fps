Set-Location "%USERPROFILE%\Documents\GitHub\pcsx2-kh2fm-60fps"
$card = "$env:APPDATA\EmuDeck\Emulators\PCSX2-Qt\memcards\Mcd002.ps2"
New-Item -ItemType Directory -Force work\saves | Out-Null
foreach ($n in "00", "01", "02") {
  python tools\memcard.py $card --extract "BISLPM-66675FM-$n/BISLPM-66675FM-$n" --out "work\saves\FM-$n.bin"
  python tools\memcard.py $card --extract "BISLPM-66675FM-$n/icon.sys" --out "work\saves\FM-$n.icon.sys"
}
$py = @'
import struct, pathlib
for n in ("00", "01", "02"):
    icon = pathlib.Path(f"work/saves/FM-{n}.icon.sys").read_bytes()
    title = icon[0xC0:0xC0 + 68].split(b"\0\0", 1)[0]
    try:
        t = title.decode("shift_jis", "replace")
    except Exception:
        t = repr(title)
    data = pathlib.Path(f"work/saves/FM-{n}.bin").read_bytes()
    print(f"== FM-{n}: icon title {t!r}  size {len(data)}")
    for off in range(0, 0x40, 16):
        print(f"   {off:04X}  " + " ".join(f"{b:02X}" for b in data[off:off + 16]))
'@
$f = "$env:TEMP\kh2_saves.py"; Set-Content -Path $f -Value $py -Encoding utf8
python $f
